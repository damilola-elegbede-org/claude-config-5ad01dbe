 def pretooluse_hooks(settings, tool):
     out = []
     for entry in settings.get("hooks", {}).get("PreToolUse", []):
         if tool not in entry.get("matcher", "").split("|"):
             continue
         for hook in entry.get("hooks", []):
             if hook.get("type") == "command" and hook.get("command"):
